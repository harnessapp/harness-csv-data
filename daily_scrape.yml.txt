name: Daily Scrape (Harness API)

on:
  workflow_dispatch: {}
  schedule:
    # 12:30am Sydney during AEDT (13:30 UTC)
    - cron: "30 13 * * *"

jobs:
  run:
    runs-on: ubuntu-latest
    timeout-minutes: 360

    steps:
      - name: Checkout harness_api
        uses: actions/checkout@v4

      - name: Set up Python
        uses: actions/setup-python@v5
        with:
          python-version: "3.11"

      - name: Install dependencies
        run: |
          python -m pip install --upgrade pip
          pip install pandas requests beautifulsoup4 numpy

      - name: Run scrapers
        run: |
          python run_scrapers.py

      - name: Upload CSVs to harness-csv-data repo
        env:
          # IMPORTANT: This must be a secret (see Part C)
          GITHUB_TOKEN: ${{ secrets.HARNESS_CSV_TOKEN }}
          REPO_OWNER: harnessapp
          REPO_NAME: harness-csv-data
          BRANCH: main

          # CRITICAL: In GitHub Actions there is NO C:\Users\joel\...
          # So we point SOURCE_DIR to wherever the files exist in this repo after scraping.
          CSV_SOURCE_DIR: .

          # merged_file.csv is expected to exist in the harness_api workspace
          MERGED_CSV_PATH: merged_file.csv
          BUILD_MERGED_META: "1"
          COMMIT_MESSAGE: "🔄 Auto-update CSVs"
        run: |
          python upload_csv_to_github.py
